15 /// @brief Which shaders draw a world_renderer_3d's meshes, and how they
16 /// filter textures. The defaults are the built-in shaders,
17 /// shaders/diffuse_vs.hlsl and shaders/diffuse_ps.hlsl. A shader is named
18 /// by its file name without .hlsl; put your own next to them in /shaders.
19 /// Your shaders get the same vertices and constants as the diffuse ones,
20 /// so start from a copy of those, which declare the first two of these:
21 /// - per draw, at register(b0): `world` (object space to world space),
22 /// the rows of the normal matrix, and the material's `lit`,
23 /// `alpha_cutoff` and `base_color`
24 /// - per frame, at register(b2): the camera's `view_projection` (world
25 /// space to the screen) and `camera_position` (in world space)
26 /// - at register(b1): the data you hand to
27 /// world_renderer_3d::set_shader_data
28 /// @code{.cpp}
29 /// buz::set_surface_shading(world, { .vertex_shader = "fog_vs", .pixel_shader = "fog_ps" });
30 /// @endcode
31 struct surface_shading
32 {
33 /// @brief The vertex shader's file name in /shaders, without ".hlsl".
34 std::string vertex_shader = "diffuse_vs";
35 /// @brief The pixel shader's file name in /shaders, without ".hlsl".
36 std::string pixel_shader = "diffuse_ps";
37 /// @brief How the meshes' textures are filtered.
38 texture_filter filter = texture_filter::linear;
39
40 bool operator==(const surface_shading& other) const = default;

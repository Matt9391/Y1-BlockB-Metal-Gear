20 /// Owned by buz::game_3d and handed to on_render_world every frame; the

22 /// farther ones whatever that order (there is a depth buffer).

24 /// void my_game::on_render_world(buz::engine& engine, buz::world_renderer_3d& world)

27 /// world.draw(*m_snake_mesh, m_snake_material, m_snake_transform);
28 /// }
29 /// @endcode
30 ///

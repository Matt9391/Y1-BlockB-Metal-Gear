51 /// @brief Draws the 3D game world, through a camera.
52 ///
53 /// Owned by buz::game_3d and handed to on_render_world every frame; the
54 /// world is drawn in the order of your calls, and nearer surfaces hide
55 /// farther ones whatever that order (there is a depth buffer).
56 /// @code{.cpp}
57 /// void my_game::on_render_world(buz::engine& engine, buz::world_renderer_3d& world)
58 /// {
59 /// world.draw_box({ 0.0f, 0.5f, 0.0f }, { 1.0f, 1.0f, 1.0f }, m_crate_material);
60 /// world.draw(*m_snake_mesh, m_snake_material, *m_snake_texture, m_snake_transform);
61 /// }
62 /// @endcode
63 ///
64 /// Until your game sets a camera, the world is seen from a fixed spot up,
65 /// behind and to the side of the origin, looking at it.

156 //----------------------------------------------------------
157 /// @brief Hands data to your own mesh shaders (see
158 /// buz::set_surface_shading), for every draw that follows, in this
159 /// frame and the next ones, until you set it again. The shaders read
160 /// it as a cbuffer at register(b1):
161 /// @code{.hlsl}
162 /// [[vk::binding(0, 2)]] cbuffer shader_data : register(b1)
163 /// {
164 /// float4 fog_color;
165 /// float fog_distance;
166 /// };
167 /// @endcode
168 /// The struct you pass and the cbuffer must agree on the layout: hlsl
169 /// packs the members in rows of four floats, and a member never
170 /// straddles two rows. The built-in shaders don't read it.
171 ///
172 /// The engine fills the other two blocks: the draw's `world` matrix,
173 /// normal matrix, `lit`, `alpha_cutoff` and `base_color` at
174 /// register(b0), and the camera's `view_projection` and
175 /// `camera_position` at register(b2), with [[vk::binding(0, 3)]]
176 /// (buz::surface_shading lists them; the diffuse shaders declare
177 /// them).
178 /// @param data At most 4096 bytes; a copy is kept.
179 void set_shader_data(std::span<const std::byte> data);

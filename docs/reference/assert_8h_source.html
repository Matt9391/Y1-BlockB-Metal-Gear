58/// @brief Halts the program in the debugger when the condition is false.
59///
60/// Use it to document assumptions your code makes:
61/// @code{.cpp}
62/// BUZ_ASSERT(frame_index < frame_count);
63/// @endcode
64/// Compiled away in release builds.
65///
66/// A macro (instead of a function) is still needed here: only a macro can
67/// stringize the condition with the # operator, skip evaluating it entirely in
68/// release, and fire BUZ_DEBUG_BREAK() at the call site instead of inside a
69/// helper function.
70#if defined(BUZ_ASSERT_ENABLED)
71 #define BUZ_ASSERT(condition) \
72 do \
73 { \
74 if (!(condition)) \
75 { \
76 buz::internal::on_assert_failed(#condition, {}); \
77 BUZ_DEBUG_BREAK(); \
78 } \
79 } while (false)
80#else
81 #define BUZ_ASSERT(condition) ((void)0)
82#endif
83
84//--------------------------------------------------------------

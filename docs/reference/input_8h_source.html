100 /// @brief The keyboard and mouse state of this frame.
101 ///
102 /// The engine owns one instance (engine::get_input), which reports the
103 /// keyboard and mouse events of this frame. Game code queries it like
104 /// this:
105 /// @code{.cpp}
106 /// if (input.is_key_down(buz::key::a)) { } // every frame while held
107 /// if (input.was_key_pressed(buz::key::space)) { } // only the frame it went down
108 /// if (input.was_key_released(buz::key::escape)) { } // only the frame it came up
109 /// @endcode
110 class input
111 {
112 //----------------------------------------------------------
113 // Lifetime
114 public:
115 //----------------------------------------------------------
116 /// @brief Reads the input the engine recorded, and locks the cursor
117 /// in the engine's window; only the engine has the recorder and the
118 /// window to pass in.
119 input(const input_recorder& recorder, native_window& window);
120
121 input(const input&) = delete;
122 input& operator=(const input&) = delete;
123
124 //----------------------------------------------------------
125 // Keyboard
126 public:
127 //----------------------------------------------------------
128 /// @brief Returns true while the key is held down.
129 bool is_key_down(key value) const;
130
131 //----------------------------------------------------------
132 /// @brief Returns true only in the frame the key went down.
133 bool was_key_pressed(key value) const;
134
135 //----------------------------------------------------------
136 /// @brief Returns true only in the frame the key came up.
137 bool was_key_released(key value) const;
138
139 //----------------------------------------------------------
140 // Mouse
141 public:
142 //----------------------------------------------------------
143 /// @brief Returns true while the mouse button is held down.
144 bool is_mouse_button_down(mouse_button button) const;
145
146 //----------------------------------------------------------
147 /// @brief Returns true only in the frame the mouse button went down.
148 bool was_mouse_button_pressed(mouse_button button) const;
149
150 //----------------------------------------------------------
151 /// @brief Returns true only in the frame the mouse button came up.
152 bool was_mouse_button_released(mouse_button button) const;
153
154 //----------------------------------------------------------
155 /// @brief Returns the mouse position in pixels, relative to the
156 /// top-left corner of the drawable area.
157 vec2 get_mouse_position() const;
158
159 //----------------------------------------------------------
160 /// @brief Returns how far the mouse wheel turned this frame:
161 /// positive away from you, negative towards you, one "click" is 1.
162 float get_mouse_wheel_delta() const;
163
164 //----------------------------------------------------------
165 /// @brief Returns how far the mouse moved this frame, in pixels:
166 /// positive x to the right, positive y down. Unlike the position, the
167 /// movement keeps changing while the cursor is locked, so it is the
168 /// value to use for mouse look:
169 /// @code{.cpp}
170 /// m_yaw -= input.get_mouse_movement().x * k_radians_per_pixel;
171 /// @endcode
172 vec2 get_mouse_movement() const;
173
174 //----------------------------------------------------------
175 // Cursor
176 public:
177 //----------------------------------------------------------
178 /// @brief Hides the cursor and keeps it inside the window, so the
179 /// mouse can turn a camera without the cursor hitting the edge of
180 /// the screen. get_mouse_position stays where it was; read
181 /// get_mouse_movement instead. Unlock it for menus.
182 void lock_cursor();
183
184 //----------------------------------------------------------
185 /// @brief Shows the cursor again and lets it leave the window.
186 void unlock_cursor();
187
188 //----------------------------------------------------------
189 /// @brief Returns true while the cursor is locked.
190 bool is_cursor_locked() const;
191
192 private:
193 const input_recorder& m_recorder; // the engine's side: what happened this frame
194 native_window& m_window; // the engine's side: the window the cursor is locked in
195 };
196}

99 /// @brief The keyboard and mouse state of this frame.
100 ///
101 /// The engine owns one instance (engine::get_input) and records every
102 /// keyboard and mouse event into it. Game code just asks questions:
103 /// @code{.cpp}
104 /// if (input.is_key_down(buz::key::a)) { } // every frame while held
105 /// if (input.was_key_pressed(buz::key::space)) { } // only the frame it went down
106 /// if (input.was_key_released(buz::key::escape)) { } // only the frame it came up
107 /// @endcode
108 class input
109 {
110 //----------------------------------------------------------
111 // Lifetime
112 public:
113 //----------------------------------------------------------
114 /// @brief Answers questions about what the engine recorded; only the
115 /// engine has one to pass in.
116 explicit input(const input_recorder& recorder);
117
118 input(const input&) = delete;
119 input& operator=(const input&) = delete;
120
121 //----------------------------------------------------------
122 // Keyboard
123 public:
124 //----------------------------------------------------------
125 /// @brief Returns true while the key is held down.
126 bool is_key_down(key value) const;
127
128 //----------------------------------------------------------
129 /// @brief Returns true only in the frame the key went down.
130 bool was_key_pressed(key value) const;
131
132 //----------------------------------------------------------
133 /// @brief Returns true only in the frame the key came up.
134 bool was_key_released(key value) const;
135
136 //----------------------------------------------------------
137 // Mouse
138 public:
139 //----------------------------------------------------------
140 /// @brief Returns true while the mouse button is held down.
141 bool is_mouse_button_down(mouse_button button) const;
142
143 //----------------------------------------------------------
144 /// @brief Returns true only in the frame the mouse button went down.
145 bool was_mouse_button_pressed(mouse_button button) const;
146
147 //----------------------------------------------------------
148 /// @brief Returns true only in the frame the mouse button came up.
149 bool was_mouse_button_released(mouse_button button) const;
150
151 //----------------------------------------------------------
152 /// @brief Returns the mouse position in pixels, relative to the
153 /// top-left corner of the drawable area.
154 vec2 get_mouse_position() const;
155
156 //----------------------------------------------------------
157 /// @brief Returns how far the mouse wheel turned this frame:
158 /// positive away from you, negative towards you, one "click" is 1.
159 float get_mouse_wheel_delta() const;
160
161 private:
162 const input_recorder& m_recorder; // the engine's side: what happened this frame
163 };

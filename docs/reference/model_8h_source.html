34 /// @brief One node of a model, as it was named and placed in Blender.
35 ///
36 /// A character made of rigid parts (the PS1 way) is a hierarchy of nodes:
37 /// pelvis -> torso -> head, upper arm -> lower arm, and so on, each part a
38 /// mesh that moves with its node. A level is a flat list of nodes too:
39 /// walls with meshes, and empty nodes whose names mark what they are
40 /// (col_* collision boxes, patrol_* points, camera_* positions).
41 struct model_node
42 {
43 /// @brief The node's name in Blender.
44 std::string name;
45 /// @brief Index of the parent node, -1 for a root.
46 int parent = -1;
47 /// @brief Where the node sits relative to its parent, as placed in the file.
48 joint_pose local_pose;
49 /// @brief The node's mesh; nullptr for an empty node.
50 std::shared_ptr<mesh> shape;
51 /// @brief How shape looks.
52 material look;
53 };
54
55 //--------------------------------------------------------------
56 /// @brief An animation, sampled into poses at a fixed frame rate: every
57 /// frame has a pose for every node of the model.
58 ///
59 /// Turning this into movement is up to you: keep the time, find the two
60 /// frames around it, blend between them, loop.
61 struct keyframe_clip
62 {
63 /// @brief The animation's name in Blender.
64 std::string name;
65 /// @brief How many frames were sampled per second.
66 float frames_per_second = 30.0f;
67 /// @brief A pose per node for every frame: frames[frame][node].
68 std::vector<std::vector<joint_pose>> frames;
69 };

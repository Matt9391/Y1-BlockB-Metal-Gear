22 /// @brief Where a node of a model sits relative to its parent node: its
23 /// translation, rotation and scale. A pose (a frame of an animation, or
24 /// get_rest_pose) is one node_transform for every node.
25 struct node_transform
26 {
27 /// @brief Where the node sits relative to its parent.
28 vec3 translation = { 0.0f, 0.0f, 0.0f };
29 /// @brief How the node is turned; the default is no rotation.
30 quat rotation = quat(1.0f, 0.0f, 0.0f, 0.0f);
31 /// @brief How much the node is scaled along each axis.
32 vec3 scale = { 1.0f, 1.0f, 1.0f };
33 };

36 /// @brief One node of a model, as it was named and placed in Blender.
37 ///
38 /// A character made of rigid parts (the PS1 way) is a hierarchy of nodes:
39 /// pelvis -> torso -> head, upper arm -> lower arm, and so on, each part a
40 /// mesh that moves with its node. A level is a flat list of nodes too:
41 /// walls with meshes, and empty nodes whose names mark what they are
42 /// (col_* collision boxes, patrol_* points, camera_* positions).
43 struct model_node
44 {
45 /// @brief The node's name in Blender.
46 std::string name;
47 /// @brief Index of the parent node, -1 for a root.
48 int parent = -1;
49 /// @brief Where the node sits relative to its parent, as placed in the file.
50 node_transform local_transform;
51 /// @brief The node's mesh; nullptr for an empty node.
52 std::shared_ptr<mesh> shape;
53 /// @brief How shape looks. Its base_texture names the image; load it
54 /// with buz::load_texture(engine, look.base_texture).
55 material look;
56 };

58 //--------------------------------------------------------------
59 /// @brief An animation, sampled into poses at a fixed frame rate: every
60 /// frame has a pose for every node of the model.
61 ///
62 /// Turning this into movement is up to you: keep the time, find the two
63 /// frames around it, blend between them, loop.
64 struct keyframe_clip
65 {
66 /// @brief The animation's name in Blender.
67 std::string name;
68 /// @brief How many frames were sampled per second.
69 float frames_per_second = 30.0f;
70 /// @brief A pose per node for every frame: frames[frame][node].
71 std::vector<std::vector<node_transform>> frames;
72 };

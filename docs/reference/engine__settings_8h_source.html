23 /// runs. They work in every game, so your game does not check them
24 /// itself; pick other keys here when you need these ones for the game.

33 /// @brief 3D only: draws every triangle's edges instead of the solid

37 /// @brief 3D only: draws every surface at its plain color, without

41 /// @brief 3D only: hides the debug lines (see debug_draw.h), and

47 /// @brief Settings the engine starts with.
48 ///
49 /// Set the ones you want to change and keep the rest; designated
50 /// initializers read well for that:
51 /// @code{.cpp}
52 /// buz::engine engine(argc, argv, { .window_title = "my game", .sampler_filter = buz::texture_filter::nearest });
53 /// @endcode
54 /// The most common ones can also be changed at launch, without
55 /// rebuilding; the command line wins over the game's settings:
56 ///
57 /// | setting | example |
58 /// |---|---|
59 /// | `-window_width`, `-window_height` | `-window_width=1920` |
60 /// | `-window_title` | `-window_title=Demo` |
61 /// | `-gpu` | `-gpu=lowest`, or `-gpu=software` for a software rasterizer |
62 /// | `-screenshot`, `-screenshot_frame` | `-screenshot=frame.png -screenshot_frame=300` |
63 ///
64 /// A setting nothing understands is logged as a warning, so a typo
65 /// shows up instead of being silently ignored.
66 struct engine_settings
67 {
68 /// @brief Width of the window's drawable area in pixels. The window
69 /// cannot be resized once it is open.
70 uint32_t window_width = 1280;
71
72 /// @brief Height of the window's drawable area in pixels.
73 uint32_t window_height = 720;
74
75 /// @brief Text in the window's title bar and on the taskbar.
76 std::string window_title = "BuzEngine";
77
78 /// @brief Which graphics card to run on; lowest_score is handy to
79 /// test how the game behaves on weaker hardware, software runs
80 /// without one.
81 adapter_preference gpu_preference = adapter_preference::highest_score;
82
83 /// @brief The background: the color every frame starts with before
84 /// anything is drawn.
85 color clear_color = { 0.1f, 0.1f, 0.15f, 1.0f };
86
87 /// @brief How textures are filtered when scaled on screen; nearest
88 /// keeps pixel art crisp instead of blurring it like linear does.
89 texture_filter sampler_filter = texture_filter::linear;
90
91 /// @brief Wait for the monitor between frames: no tearing, and the
92 /// frame rate stays at the monitor's refresh rate.
93 vsync use_vsync = vsync::yes;
94
95 /// @brief Show the stats overlay (frame rate, draw calls) from the
96 /// start. A debug key switches it while the game runs.
97 stats_overlay show_stats = stats_overlay::no;
98
99 /// @brief The keys that switch the debug views while the game runs.
100 debug_keys debug_controls;
101
102 /// @brief When set, the engine saves frame screenshot_frame as a png
103 /// at this path and quits. Until then every frame lasts exactly
104 /// 1/60 s and the keyboard and mouse are ignored, so every run, and
105 /// every graphics api, draws the very same frame: the way to compare
106 /// two versions pixel by pixel. Relative paths end up in the game's
107 /// `saved/<game>` folder next to the executable.
108 std::string screenshot_path;
109
110 /// @brief The frame a screenshot run saves (see screenshot_path),
111 /// counted from 0.
112 int screenshot_frame = 120;
113 };

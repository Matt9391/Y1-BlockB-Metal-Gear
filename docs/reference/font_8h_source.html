16 /// @brief A fixed-grid bitmap font: one atlas texture holding every
17 /// character in a regular grid, in ascii order.
18 ///
19 /// All glyph data lives in the atlas image (content/fonts/), this struct
20 /// only describes the grid so draw_text can find each character:
21 /// @code{.cpp}
22 /// buz::font font;
23 /// font.atlas = buz::load_texture(engine, "fonts/font_8x8.png");
24 /// renderer.draw_text(font, "hello", { 20.0f, 20.0f }, 16.0f);
25 /// @endcode
26 ///
27 /// The defaults match the engine's built-in atlas
28 /// (content/fonts/font_8x8.png): 8x8 pixel glyphs, 16 per row, starting
29 /// at the space character (ascii 32).
30 struct font
31 {
32 /// @brief The texture with every glyph.
33 std::shared_ptr<texture> atlas;
34 /// @brief The character of the first glyph, in ascii.
35 uint32_t first_character = 32;
36 /// @brief How many glyphs the atlas holds.
37 uint32_t glyph_count = 96;
38 /// @brief How many glyphs one row of the atlas holds.
39 uint32_t glyphs_per_row = 16;
40 /// @brief Size of one glyph in pixels.
41 uvec2 glyph_size = { 8, 8 };
42
43 //----------------------------------------------------------
44 /// @brief Returns where a character's glyph sits in the atlas, in
45 /// pixels. Characters the font does not have come out as the first
46 /// glyph (usually space, so they show up as blank).

48 };

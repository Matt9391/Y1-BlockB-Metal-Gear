22 /// @brief Owns all engine systems and runs the main loop.
23 ///
24 /// The engine comes first, the game second: the game needs the engine to
25 /// load its textures, so it takes it in its constructor. C++ destroys
26 /// local variables in reverse order, so the game, and every texture and
27 /// mesh it holds, is gone before the engine shuts the graphics card down.
28 /// @code{.cpp}
29 /// int main(int argc, char** argv)
30 /// {
31 /// buz::engine engine(argc, argv, { .window_title = "my game" });
32 /// my_game game(engine);
33 /// return engine.run(game);
34 /// }
35 /// @endcode

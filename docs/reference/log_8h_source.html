10// Logging is compiled in for Debug and Development builds. In Release, the
11// shipping build (NDEBUG without BUZ_DEVELOPMENT), every log call collapses
12// to nothing, so no formatting cost is paid and no log file is created. The
13// format string is still checked at compile time in both builds:
14// std::format_string checks the placeholders against the argument types
15// while compiling, so a typo like log::info("{} {}", one_value) is a compile
16// error. A format string that is only known at runtime needs std::vformat
17// instead.
18#if !defined(NDEBUG) || defined(BUZ_DEVELOPMENT)
19 #define BUZ_LOG_ENABLED
20#endif
